async function updateInfo(){ const last = document.getElementById('last'); const state = document.getElementById('updateState'); state.className = 'update-state'; state.textContent = 'در حال بررسی اطلاعات آنلاین…'; try { const response = await fetch('./data/events.json?t=' + Date.now(), { cache: 'no-store' }); if (!response.ok) { throw new Error('HTTP ' + response.status); } const data = await response.json(); if (data.events) { const dynamicEvents = {}; Object.values(data.events).forEach(item => { const title = item.title || ''; const link = item.link || ''; const date = item.date || ''; let category = 'رویدادهای خاص'; if ( title.includes('کنسرت') || title.includes('موسیقی') || title.includes('خواننده') || title.includes('ارکستر') ) { category = 'موسیقی'; } else if ( title.includes('تئاتر') || title.includes('نمایش') ) { category = 'تئاتر'; } else if ( title.includes('نمایشگاه') || title.includes('گالری') || title.includes('هنر') ) { category = 'نمایشگاه‌ها'; } if (!dynamicEvents[category]) { dynamicEvents[category] = []; } dynamicEvents[category].push([ title, date, category === 'موسیقی' ? 'موسیقی' : category === 'تئاتر' ? 'تئاتر' : category === 'نمایشگاه‌ها' ? 'نمایشگاه هنری' : 'رویداد', 'اطلاعات آنلاین', 'رویداد دریافت‌شده از منبع آنلاین', link, 'به‌روزرسانی خودکار' ]); }); if (Object.keys(dynamicEvents).length > 0) { events.mehr['رویدادهای آنلاین'] = []; Object.values(dynamicEvents).forEach(items => { items.forEach(item => { events.mehr['رویدادهای آنلاین'].push(item); }); }); } render(); renderAll(); } const updateDate = data.updated_at ? new Date(data.updated_at) : new Date(); last.textContent = 'آخرین به‌روزرسانی: ' + new Intl.DateTimeFormat('fa-IR', { dateStyle: 'full', timeStyle: 'short' }).format(updateDate); state.classList.add('update-online'); state.textContent = 'اطلاعات آنلاین با موفقیت دریافت و اعمال شد.'; } catch (error) { console.error(error); state.classList.add('update-error'); state.textContent = 'دریافت اطلاعات آنلاین ناموفق بود؛ اطلاعات فعلی حفظ شد.'; } }
